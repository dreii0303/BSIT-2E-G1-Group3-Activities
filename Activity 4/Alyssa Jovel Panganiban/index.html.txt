<script>
    const form = document.getElementById('dataForm');
    const resultCard = document.getElementById('result-card');
    const resultContent = document.getElementById('result-content');
    const errorMessage = document.getElementById('error-message');

    form.addEventListener('submit', function(event) {
    event.preventDefault();

    const fullname = document.getElementById('fullname').value.trim();
    const gender = document.getElementById('gender').value;
    const dob = document.getElementById('dob').value;
    const status = document.getElementById('status').value;
    const email = document.getElementById('email').value.trim();
    const mobile = document.getElementById('mobile').value.trim();
    const address = document.getElementById('address').value.trim();

    
    if (fullname === '' && gender === '') {
        errorMessage.textContent = "Please enter your Full Name and select Gender.";
        errorMessage.style.display = 'block';
        resultCard.style.display = 'none';
        return;
    }
    if (fullname === '') {
        errorMessage.textContent = "Please enter your Full Name.";
        errorMessage.style.display = 'block';
        resultCard.style.display = 'none';
        return;
    }
    if (gender === '') {
        errorMessage.textContent = "Please select Gender.";
        errorMessage.style.display = 'block';
        resultCard.style.display = 'none';
        return;
    }

    
    if (dob === '') {
        errorMessage.textContent = "Please select your Date of Birth.";
        errorMessage.style.display = 'block';
        resultCard.style.display = 'none';
        return;
    }

    
    if (status === '') {
        errorMessage.textContent = "Please select your Civil Status.";
        errorMessage.style.display = 'block';
        resultCard.style.display = 'none';
        return;
    }

    
    if (email === '') {
        errorMessage.textContent = "Please enter your Email Address.";
        errorMessage.style.display = 'block';
        resultCard.style.display = 'none';
        return;
    }
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
        errorMessage.textContent = "Please enter a valid Email Address.";
        errorMessage.style.display = 'block';
        resultCard.style.display = 'none';
        return;
    }

    if (mobile === '') {
        errorMessage.textContent = "Please enter your Mobile Number.";
        errorMessage.style.display = 'block';
        resultCard.style.display = 'none';
        return;
    }
    if (!/^\d{10,15}$/.test(mobile)) {
        errorMessage.textContent = "Please enter a valid Mobile Number (10-15 digits).";
        errorMessage.style.display = 'block';
        resultCard.style.display = 'none';
        return;
    }

    if (address === '') {
        errorMessage.textContent = "Please enter your Complete Address.";
        errorMessage.style.display = 'block';
        resultCard.style.display = 'none';
        return;
    }

   
    errorMessage.style.display = 'none';

    resultContent.innerHTML = `
        <div class="result-item"><strong>Full Name:</strong> ${fullname}</div>
        <div class="result-item"><strong>Date of Birth:</strong> ${dob}</div>
        <div class="result-item"><strong>Gender:</strong> ${gender}</div>
        <div class="result-item"><strong>Status:</strong> ${status}</div>
        <div class="result-item"><strong>Email:</strong> ${email}</div>
        <div class="result-item"><strong>Mobile:</strong> ${mobile}</div>
        <div class="result-item"><strong>Address:</strong> ${address}</div>
    `;

    resultCard.style.display = 'block';
});

</script>
